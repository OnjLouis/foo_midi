cmake_minimum_required(VERSION 3.20)
project(external_midi_regression LANGUAGES CXX)
enable_testing()
add_executable(external-midi-test TimelineTest.cpp)
target_compile_features(external-midi-test PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT MSVC)
    target_link_options(external-midi-test PRIVATE -static)
endif()
add_test(NAME external-midi-regression COMMAND external-midi-test)
if(WIN32 AND MSVC)
    add_executable(external-midi-output-test OutputTest.cpp ../../Players/ExternalMIDI/Output.cpp)
    target_compile_features(external-midi-output-test PRIVATE cxx_std_17)
    target_compile_definitions(external-midi-output-test PRIVATE _WINMM_ NOMINMAX)
    add_test(NAME external-midi-output-ownership COMMAND external-midi-output-test)
endif()
