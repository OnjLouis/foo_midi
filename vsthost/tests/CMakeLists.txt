cmake_minimum_required(VERSION 3.20)
project(vst_port_regression LANGUAGES CXX RC)
enable_testing()
set(vst_sdk "${CMAKE_CURRENT_SOURCE_DIR}/../../3rdParty/vstsdk")
add_executable(port-host WIN32 ../vsthost.cpp Host.rc)
target_include_directories(port-host PRIVATE "${vst_sdk}")
target_compile_definitions(port-host PRIVATE UNICODE _UNICODE)
target_compile_features(port-host PRIVATE cxx_std_17)
target_link_libraries(port-host PRIVATE ole32 comctl32)
add_library(port-probe SHARED PortProbe.cpp)
target_include_directories(port-probe PRIVATE "${vst_sdk}")
target_compile_features(port-probe PRIVATE cxx_std_17)
set_target_properties(port-probe PROPERTIES PREFIX "")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND NOT MSVC)
    target_link_options(port-host PRIVATE -static)
    target_link_options(port-probe PRIVATE -static)
    target_compile_options(port-host PRIVATE -Wno-unknown-pragmas)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME vst-port-regression COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_ports.py" "$<TARGET_FILE:port-host>" "$<TARGET_FILE:port-probe>")
